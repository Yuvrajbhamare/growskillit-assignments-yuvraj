package assignmentsOnly;

public class Assignment3_1 {

	public static void main(String[] args) {
		// Count Digits

		// input 987654
		// Number of digits = 6

		int num = 987654;
		int count = 0;

		while (num > 0) {
			num = num / 10;
			count++;
		}

		System.out.println("Number of digits = " + count);
	}
}
