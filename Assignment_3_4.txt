package assignmentsOnly;

public class Assignment3_4 {

	public static void main(String[] args) {

		System.out.println("Even numbers:");

		for (int i = 1; i <= 20; i++) {
			if (i % 2 == 0) {
				System.out.print(i + " ");
			}
		}

		System.out.println("\n");

		System.out.println("Odd numbers:");

		for (int i = 1; i <= 20; i++) {
			if (i % 2 != 0) {
				System.out.print(i + " ");
			}
		}
	}
}
