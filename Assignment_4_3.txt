package assignmentsOnly;

public class Assignment4_3 {

	public static void main(String[] args) {

		int[] arr = { 1, 2, 3, 5, 6, 7, 8, 9, 10 };

		int total = 0;
		int sum = 0;

		for (int i = 1; i <= 10; i++) {
			total = total + i;
		}

		for (int i = 0; i < arr.length; i++) {
			sum = sum + arr[i];
		}

		int missing = total - sum;

		System.out.println("Missing number: " + missing);
	}
}
